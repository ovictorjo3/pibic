\section{Fotoquímica e Luminescência Atmosférica}

	\begin{frame}{Introdução}
		A atmosfera superior é uma fonte permanente de emissão de fótons que
		são liberados por átomos (moléculas) excitadas acima de seu nível normal de
		energia. A produção desta luminescência é devida a diversos processos físico-químicos e, em geral, ocorre através da emissão de linhas espectrais discretas. (pg. 53)
	\end{frame}
	
	\begin{frame}{Luminescência atmosférica}
		\begin{itemize}
			\justifying
			\item \textbf{airglow} é em geral muito fraca para ser visível a olho nu;
			\item \textbf{dayglow} luminescência atmosférica durante o dia, é mais difícil de ser medida, por causa da presença de luz solar direta e do espelhamento \textit{Rayleigh};
			\item \textbf{nightglow} luminescência atmosférica durante a noite. 
		\end{itemize}
	\end{frame}
	
	\begin{frame}{Luminescência - aurora}
		\begin{itemize}
			\item Predomina nas grandes latitudes e se distribui numa região de formato oval em torno de 67\degree de latitude geomagnética. é visível a olho nu;
			
			\item Durante as grandes tempestades magnéticas, as auroras são visíveis também em latitudes mais baixas;
		\end{itemize}
	\end{frame}
	
	\begin{frame}{Luminescência - aurora}
		\begin{itemize}
			\item Durante a atividade magnética intensa, a oval em geral se torna mais larga.
			
			\begin{center}
				\includegraphics[scale=0.25]{imagens/aurora.png}
			\end{center}
		\end{itemize}
	\end{frame}
	
	\begin{frame}{Processos de excitação/emissão}
		\begin{itemize}
			\justifying
			\item \textbf{Espelhamento ressonante} é o processo em que os fótons são absorvidos e reemetidos na frequência de ressonância do átomo (molécula). O fóton emitido tem a mesma frequência do fóton incidente;
			
			\item \textbf{Emissão fluorescente/fluorescência} o fóton emitido é de energia menor que o incidente;
			
			\begin{center}
				\includegraphics[scale=0.35]{imagens/ressonancia.png}
			\end{center}
		\end{itemize}
	\end{frame}
	
	\begin{frame}{Processos de excitação/emissão}
		\begin{itemize}
			\justifying
			\item  A \textbf{quimiluminescência} resulta das reações químicas que, sendo exotérmicas, podem deixar os produtos da reação em estados excitados, capazes de emitir fótons.
		\end{itemize}
	\end{frame}

	\begin{frame}{Fotoquímica}
		\begin{itemize}
			\justifying
			\item A distribuição de alguns componentes atmosféricos é afetada por reações químicas; 
			\item Através das reações fotoquímicas , portanto, estabelece-se uma perda ou uma produção de determinado componente atmosférico;
			\item O estudo e a detemrinação das taxas de reação são o objeto da química cinética;
			\item Constante de tempo pequena implica num processo rápido de perdao ou produção, e diz-se que a reação é rápida.
		\end{itemize}
	\end{frame}

\section{A Formação de Camadas Ionizadas}
	\begin{frame}{Introdução - Características e Nomenclatura}
		\begin{itemize}
			\justifying
			\item O principal processo de produção iônica é a absorção da radiação solar na faixa espectral do extremo ultra-violeta (EUV), e dos raios X.
			\item Em condições especiais, partículas ionizadas de energia cinética elevada também conseguem produzir ionização através de colisões com partículas neutras.
			\item Nesse caso, a ionização é caracterizada por secundária ou corpuscular.
		\end{itemize}
	\end{frame}
	
	\begin{frame}{Introdução - Características e Nomenclatura}
		\begin{itemize}
			\justifying
			\item Do ponto de vista qualitativo, é fácil notar que a produção de pares íons-elétrons deve produzir um pico na atmosfera.
			\item O número de íons produzidos é proporcional ao número de fótons incidentes e ao número de partículas ionizáveis.
			\item A densidade atmosférica diminui com altura, de modo que a absorção de fótons no topo é mínima.
		\end{itemize}
	\end{frame}

\section{As Regiões Ionosféricas}
	\begin{frame}{Região D}
		\begin{itemize}
			\justifying
			\item É a parte inferior da ionosfera, até cerca de 80 km; e é a região menos regular do que a ionosfera superior;
			\item As radiações ionizantes mais importantes são raios-X e a radiação Lyman-$\alpha$, que ioniza o óxido nítrico {NO};
			\item As reações químicas mais importantes incluem os componentes O, $O_{3}$, NO, $NO_{2}$, $CO_{2}$ e $H_{2}O$.
		\end{itemize}
	\end{frame}
	
	\begin{frame}{Região D}
		\centering
		\includegraphics[scale=0.35]{imagens/regiao_D.png}
	\end{frame}

	\begin{frame}{Região E}
		\begin{itemize}
			\justifying
			\item É a região que fica entre a D e F;
			\item Aproximadamente entre 80 e 130 km de altura;
			\item É importante pela presença de correntes elétricas que nela fluem e sua interação com o campo magnético.
		\end{itemize}
	\end{frame}
	
	\begin{frame}{A região F1}
		\begin{itemize}
			\item Ocorre a transição entre as leis quadráticas e linear para a perda eletrônica; \\

	\begin{frame}{A região F2}
		\begin{itemize}
			\justifying
			\item Engloba toda a região superior da ionosfera, inclusive a região de pico;
			\item Acima desta, a ionosfera encontra-se em equilíbrio difusivo;
			\item Isto é, o plasma se distribui com a sua própria escala de altura.
		\end{itemize}
